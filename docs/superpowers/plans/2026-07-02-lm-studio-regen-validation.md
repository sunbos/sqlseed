```python
"""Clear all table data in complex_biz.db and hr_biz.db (preserve schema).

Deletes rows in FK-reverse-topological order and resets sqlite_sequence
so AUTOINCREMENT restarts from 1. Schema objects (tables, indexes, CHECK
constraints, FK definitions) are preserved.
"""
from __future__ import annotations

import sqlite3
import sys
from pathlib import Path

# FK-reverse-topological order: children first, parents last.
# Derived from PRAGMA foreign_key_list per table; verified manually against
# the DDL of both databases.
CLEAR_ORDER: dict[str, list[str]] = {
    "complex_biz.db": [
        "order_items",
        "orders",
        "sales",
        "products",
        "items",
        "users",
        "merchants",
        "categories",
    ],
    "hr_biz.db": [
        "tasks",
        "projects",
        "employees",
        "departments",
    ],
}


def clear_db(db_path: str, tables: list[str]) -> None:
    if not Path(db_path).exists():
        print(f"SKIP {db_path}: file not found")
        return
    conn = sqlite3.connect(db_path)
    conn.execute("PRAGMA foreign_keys = OFF")
    cur = conn.cursor()
    print(f"=== Clearing {db_path} ===")
    for table in tables:
        before = cur.execute(f"SELECT COUNT(*) FROM {table}").fetchone()[0]
        cur.execute(f"DELETE FROM {table}")
        after = cur.execute(f"SELECT COUNT(*) FROM {table}").fetchone()[0]
        print(f"  {table}: {before} -> {after}")
    # Reset AUTOINCREMENT counter so next insert starts at 1
    try:
        cur.execute("DELETE FROM sqlite_sequence")
        print(f"  sqlite_sequence: reset")
    except sqlite3.OperationalError:
        pass  # no AUTOINCREMENT tables
    conn.commit()
    conn.close()


def main() -> int:
    for db_path, tables in CLEAR_ORDER.items():
        clear_db(db_path, tables)
    # Verify all tables are empty
    print("\n=== Verification ===")
    all_empty = True
    for db_path, tables in CLEAR_ORDER.items():
        conn = sqlite3.connect(db_path)
        for table in tables:
            n = conn.execute(f"SELECT COUNT(*) FROM {table}").fetchone()[0]
            status = "OK" if n == 0 else "FAIL"
            if n != 0:
                all_empty = False
            print(f"  [{status}] {db_path}.{table}: {n} rows")
        conn.close()
    print()
    if all_empty:
        print("=== RESULT: PASS — all tables cleared ===")
        return 0
    print("=== RESULT: FAIL — some tables still have rows ===")
    return 1


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Generate _e2e_generation_report.md part 1: row counts + verify_business_logic."""
from __future__ import annotations

import sqlite3
import subprocess
from pathlib import Path

DBS = ["complex_biz.db", "hr_biz.db"]


def row_counts(db_path: str) -> list[tuple[str, int]]:
    conn = sqlite3.connect(db_path)
    tables = [
        r[0]
        for r in conn.execute(
            "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name"
        ).fetchall()
    ]
    counts = [(t, conn.execute(f"SELECT COUNT(*) FROM {t}").fetchone()[0]) for t in tables]

        lines.append(f"### {db}")
        lines.append("")

        for t, n in row_counts(db):
            status = "OK" if n > 0 else "FAIL"
            lines.append(f"| {t} | {n} ({status}) |")
        lines.append("")

    Path("_e2e_generation_report.md").write_text("\n".join(lines), encoding="utf-8")
    print("Wrote _e2e_generation_report.md (row counts section)")


if __name__ == "__main__":
    main()
```

## Task 6: Validation — Cross-Column CHECK Pattern Detection

**Files:**
- Create: `_verify_cross_column_checks.py`

- [ ] **Step 1: Write the cross-column CHECK pattern detector**

Create `_verify_cross_column_checks.py`:

```python
"""Detect cross-column CHECK constraint patterns and verify YAML config coverage.

Pattern-based, schema-agnostic: works for any SQLite database.
Does NOT hardcode table or column names — derives all patterns from DDL.

Patterns detected:
  A: CHECK(col_x >= col_y)      — col_x should use derive_from=[col_y]
  B: CHECK(col_x <= col_y)      — col_x should use derive_from=[col_y]
  C: CHECK(col_x IN (lit,...))  — col_x should use choice/weighted_choice
  D: CHECK(col_x >= N AND col_x <= M)  — col_x should have min_value/max_value
"""
from __future__ import annotations

import re
import sqlite3
import sys
from pathlib import Path
from typing import Any

import yaml


# Regex patterns for cross-column CHECK constraints (schema-agnostic).
# These match any column reference on either side, not specific names.
CROSS_COL_PATTERNS: list[tuple[str, str, re.Pattern[str]]] = [
    # Pattern A: col_x >= col_y (both columns, no literals)
    ("A", "col_x >= col_y", re.compile(r"CHECK\s*\(\s*(\w+)\s*>=\s*(\w+)\s*\)", re.IGNORECASE)),
    # Pattern B: col_x <= col_y (both columns, no literals)
    ("B", "col_x <= col_y", re.compile(r"CHECK\s*\(\s*(\w+)\s*<=\s*(\w+)\s*\)", re.IGNORECASE)),
    # Pattern D: col_x >= N AND col_x <= M (range with literals)
    (
        "D",
        "col_x in [N, M]",
        re.compile(
            r"CHECK\s*\(\s*(\w+)\s*>=\s*(\d+)\s+AND\s+\1\s*<=\s*(\d+)\s*\)",
            re.IGNORECASE,
        ),
    ),
]

# Pattern C: col_x IN (literal, ...) — enum constraint
ENUM_PATTERN: re.Pattern[str] = re.compile(
    r"CHECK\s*\(\s*(\w+)\s+IN\s*\(([^)]+)\)\s*\)", re.IGNORECASE
)

# Literal detection: a "column" is actually a literal if it's a number or quoted string.
LITERAL_RE = re.compile(r"^\d+(\.\d+)?$|^[\"'].*[\"']$")


def is_literal(token: str) -> bool:
    """Return True if token is a numeric or quoted-string literal."""
    return bool(LITERAL_RE.match(token.strip()))


def get_table_ddls(db_path: str) -> list[tuple[str, str]]:
    """Return [(table_name, ddl), ...] for all user tables."""
    conn = sqlite3.connect(db_path)
    rows = conn.execute(
        "SELECT name, sql FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name"
    ).fetchall()
    conn.close()
    return [(r[0], r[1] or "") for r in rows]


def detect_patterns(ddl: str) -> list[dict[str, Any]]:
    """Detect all cross-column CHECK patterns in a table DDL.

    Returns a list of dicts: {pattern_id, description, col_x, col_y, col_y_is_literal}
    """
    patterns: list[dict[str, Any]] = []
    for pid, desc, regex in CROSS_COL_PATTERNS:
        for m in regex.finditer(ddl):
            col_x = m.group(1)
            col_y = m.group(2) if pid in ("A", "B") else None
            # For pattern A/B: skip if col_y is a literal (that's pattern D-ish)
            if pid in ("A", "B") and col_y and is_literal(col_y):
                continue
            patterns.append(
                {
                    "pattern_id": pid,
                    "description": desc,
                    "col_x": col_x,
                    "col_y": col_y,
                    "literal_low": m.group(2) if pid == "D" else None,
                    "literal_high": m.group(3) if pid == "D" else None,
                }
            )
    # Pattern C: enum
    for m in ENUM_PATTERN.finditer(ddl):
        col_x = m.group(1)
        literals = [s.strip().strip("'\"") for s in m.group(2).split(",")]
        patterns.append(
            {
                "pattern_id": "C",
                "description": "col_x IN (literals)",
                "col_x": col_x,
                "literals": literals,
            }
        )
    return patterns


def load_yaml_config(yaml_path: str) -> dict[str, Any]:
    """Load YAML config, indexed by table name -> column name -> config."""
    if not Path(yaml_path).exists():
        return {}
    with open(yaml_path, encoding="utf-8") as f:
        data = yaml.safe_load(f) or {}
    # data shape: {tables: [{name, columns: [{name, generator, params, derive_from, ...}]}]}
    indexed: dict[str, dict[str, dict[str, Any]]] = {}
    for table in data.get("tables", []):
        tname = table.get("name", "")
        cols: dict[str, dict[str, Any]] = {}
        for col in table.get("columns", []):
            cname = col.get("name", "")
            if cname:
                cols[cname] = col
        indexed[tname] = cols
    return indexed


def check_pattern_coverage(
    table: str,
    pattern: dict[str, Any],
    table_cols: dict[str, dict[str, Any]],
) -> dict[str, Any]:
    """Check if the YAML config covers a detected pattern for the table."""
    col_x = pattern["col_x"]
    col_cfg = table_cols.get(col_x, {})
    generator = col_cfg.get("generator")
    derive_from = col_cfg.get("derive_from")
    params = col_cfg.get("params") or {}

    pid = pattern["pattern_id"]
    if pid in ("A", "B"):
        # Expect derive_from=[col_y] for col_x
        col_y = pattern["col_y"]
        covered = bool(
            derive_from
            and isinstance(derive_from, list)
            and col_y in derive_from
        )
        return {
            "pattern": pid,
            "table": table,
            "col_x": col_x,
            "col_y": col_y,
            "expected": f"derive_from=[{col_y}]",
            "actual": f"derive_from={derive_from}, generator={generator}",
            "covered": covered,
        }
    if pid == "C":
        # Expect generator=choice or weighted_choice
        covered = generator in ("choice", "weighted_choice")
        return {
            "pattern": pid,
            "table": table,
            "col_x": col_x,
            "expected": "generator in (choice, weighted_choice)",
            "actual": f"generator={generator}",
            "covered": covered,
        }
    if pid == "D":
        # Expect min_value/max_value in params matching literals
        low = int(pattern["literal_low"])
        high = int(pattern["literal_high"])
        actual_low = params.get("min_value")
        actual_high = params.get("max_value")
        covered = actual_low == low and actual_high == high
        return {
            "pattern": pid,
            "table": table,
            "col_x": col_x,
            "expected": f"params.min_value={low}, params.max_value={high}",
            "actual": f"params.min_value={actual_low}, params.max_value={actual_high}",
            "covered": covered,
        }
    return {"pattern": pid, "table": table, "covered": False, "error": "unknown pattern"}


def main() -> int:
    pairs = [
        ("complex_biz.db", "complex_biz_config.yaml"),
        ("hr_biz.db", "hr_biz_config.yaml"),
    ]
    all_results: list[dict[str, Any]] = []
    for db_path, yaml_path in pairs:
        print(f"=== {db_path} (config: {yaml_path}) ===")
        ddls = get_table_ddls(db_path)
        yaml_cfg = load_yaml_config(yaml_path)
        for table, ddl in ddls:
            patterns = detect_patterns(ddl)
            if not patterns:
                continue
            table_cols = yaml_cfg.get(table, {})
            for p in patterns:
                result = check_pattern_coverage(table, p, table_cols)
                result["db"] = db_path
                all_results.append(result)
                status = "COVERED" if result["covered"] else "GAP"
                print(f"  [{status}] Pattern {result['pattern']} {table}.{result.get('col_x', '?')}")
                print(f"          expected: {result['expected']}")
                print(f"          actual:   {result['actual']}")

    # Summary
    total = len(all_results)
    covered = sum(1 for r in all_results if r["covered"])
    gaps = total - covered
    print()
    print(f"=== SUMMARY: {covered}/{total} patterns covered, {gaps} gaps ===")

    # Append to report
    with Path("_e2e_generation_report.md").open("a", encoding="utf-8") as f:
        f.write("\n## 3. Cross-Column CHECK Pattern Coverage\n\n")
        f.write("| DB | Pattern | Table | Col | Expected | Actual | Status |\n")
        f.write("|----|---------|-------|-----|----------|--------|--------|\n")
        for r in all_results:
            status = "COVERED" if r["covered"] else "GAP"
            f.write(
                f"| {r['db']} | {r['pattern']} | {r['table']} | "
                f"{r.get('col_x', '?')} | {r['expected']} | {r['actual']} | {status} |\n"
            )
        f.write(f"\n**Total: {covered}/{total} covered, {gaps} gaps**\n")
    print("Appended cross-column pattern section to _e2e_generation_report.md")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
